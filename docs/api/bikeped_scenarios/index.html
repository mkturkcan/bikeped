    builders ¶

    Path builders that emit the testbench waypoint format.

    Every builder works in the scenario frame used by decision_testbench: metres, x along the road (bikes ride along it), y across the road (pedestrians cross along it), crosswalk centre at (5, 0). Times are seconds from the start of the scenario and speeds are metres per second.

    The testbench interpolates paths linearly in time, so curved or speed-varying motion is sampled densely (every step seconds, 0.25 s by default). Headings are in degrees, counter-clockwise from +x: 0 rides towards +x, 180 towards -x (the direction of the conformance bikes), 90 walks towards +y.

    The fluent :class:Path keeps scenario definitions short::

    bike = (Path.start(34, 1.5, heading=180).ride(14, 6.0)
        .arc(radius=6, angle=90).ride(8).build())

    def ride(self, distance: float, speed: float | None = None) -> Path:
    """Move ``distance`` metres along the current heading at constant ``speed`` (m/s)."""
    if speed is not None:
        self.speed = float(speed)
    if self.speed <= 0:
        raise ValueError('ride() needs a positive speed')
    ux, uy = self._unit()
    x0, y0, t0 = self.x, self.y, self.t
    dur = distance / self.speed
    n = max(1, math.ceil(dur / self.step))
    for i in range(1, n + 1):
        f = i / n
        self._emit(x0 + ux * distance * f, y0 + uy * distance * f, t0 + dur * f)
    return self

    def to(self, x: float, y: float, speed: float | None = None) -> Path:
    """Move in a straight line to ``(x, y)`` at constant ``speed``; the heading follows."""
    dx, dy = x - self.x, y - self.y

    if dist < 1e-9:

    self.heading = math.degrees(math.atan2(dy, dx))
    return self.ride(dist, speed)

    def change_speed(self, v_end: float, accel: float) -> Path:
    """Accelerate or brake along the heading to ``v_end`` m/s at ``|accel|`` m/s^2.

    The distance covered follows from constant acceleration; the path ends with
    ``speed == v_end``. Braking to zero leaves the agent stopped at the end point.
    """
    a = abs(float(accel))
    v0 = self.speed
    if a <= 0 or abs(v_end - v0) < 1e-9:
        return self
    sign = 1.0 if v_end > v0 else -1.0
    dur = abs(v_end - v0) / a
    ux, uy = self._unit()
    x0, y0, t0 = self.x, self.y, self.t
    n = max(1, math.ceil(dur / self.step))
    for i in range(1, n + 1):
        tt = dur * i / n
        s = v0 * tt + 0.5 * sign * a * tt * tt
        self._emit(x0 + ux * s, y0 + uy * s, t0 + tt)
    self.speed = float(v_end)
    return self

    def brake_to(self, x: float | None = None, y: float | None = None,
             v_end: float = 0.0) -> Path:
    """Brake at constant deceleration so the agent reaches ``v_end`` at a given coordinate.

    Give ``x`` (for agents moving along the road) or ``y`` (for agents moving across it). The
    deceleration is whatever stops the agent exactly there.
    """
    ux, uy = self._unit()
    if x is not None:
        dist = (x - self.x) / ux if abs(ux) > 1e-6 else math.nan
    elif y is not None:
        dist = (y - self.y) / uy if abs(uy) > 1e-6 else math.nan
    else:
        raise ValueError('brake_to() needs x or y')
    if not dist > 0:
        raise ValueError('brake_to() target lies behind the agent')
    decel = (self.speed ** 2 - v_end ** 2) / (2.0 * dist)
    if decel <= 0:
        raise ValueError('brake_to() needs v_end below the current speed')
    return self.change_speed(v_end, decel)

    def arc(self, radius: float, angle: float, speed: float | None = None) -> Path:
    """Turn on a circular arc of ``radius`` metres through ``angle`` degrees.

    Positive angles turn left (counter-clockwise seen from above, ``+y`` up), negative angles
    turn right. Speed is constant along the arc.
    """
    if speed is not None:
        self.speed = float(speed)
    if self.speed <= 0:
        raise ValueError('arc() needs a positive speed')
    sgn = 1.0 if angle >= 0 else -1.0
    h0 = math.radians(self.heading)
    cx = self.x - sgn * radius * math.sin(h0)
    cy = self.y + sgn * radius * math.cos(h0)
    length = radius * math.radians(abs(angle))
    dur = length / self.speed
    n = max(2, math.ceil(dur / self.step))
    t0 = self.t
    for i in range(1, n + 1):
        phi = h0 + sgn * math.radians(abs(angle)) * i / n
        self._emit(cx + sgn * radius * math.sin(phi), cy - sgn * radius * math.cos(phi),
                   t0 + dur * i / n)
    self.heading += angle
    return self

    def lane_change(self, lateral: float, length: float, speed: float | None = None) -> Path:
    """Shift sideways by ``lateral`` metres (positive to the left) over ``length`` metres.

    The lateral offset follows a half-cosine, which gives a smooth swerve or lane change; the
    heading is unchanged at the end.
    """
    if speed is not None:
        self.speed = float(speed)
    ux, uy = self._unit()
    lx, ly = -uy, ux
    x0, y0, t0 = self.x, self.y, self.t
    dur = math.hypot(length, lateral) / self.speed
    n = max(4, math.ceil(dur / self.step))
    for i in range(1, n + 1):
        f = i / n
        off = lateral * 0.5 * (1.0 - math.cos(math.pi * f))
        self._emit(x0 + ux * length * f + lx * off, y0 + uy * length * f + ly * off,
                   t0 + dur * f)
    return self

    def zigzag(self, distance: float, amplitude: float, wavelength: float,

    def through(self, points: Sequence[tuple[float, float]], speed: float | None = None) -> Path:
    """Follow a Catmull-Rom spline from the current position through ``points`` at ``speed``.

    The spline passes through every control point. Arc length is resampled so the speed is
    constant; the heading at the end follows the last segment.
    """
    if speed is not None:
        self.speed = float(speed)
    xy = catmull_rom([(self.x, self.y), *points], samples_per_segment=24)
    seg = np.hypot(np.diff(xy[:, 0]), np.diff(xy[:, 1]))
    s = np.concatenate([[0.0], np.cumsum(seg)])
    total = float(s[-1])
    dur = total / self.speed
    n = max(2, math.ceil(dur / self.step))
    t0 = self.t
    for i in range(1, n + 1):
        si = total * i / n
        self._emit(float(np.interp(si, s, xy[:, 0])), float(np.interp(si, s, xy[:, 1])),
                   t0 + dur * i / n)
    dx, dy = xy[-1] - xy[-2]
    self.heading = math.degrees(math.atan2(dy, dx))
    return self

    def catmull_rom(points: Sequence[tuple[float, float]], samples_per_segment: int = 16) -> np.ndarray:
    """Sample a centripetal-free (uniform) Catmull-Rom spline through ``points``.

    Returns an ``(N, 2)`` array in the same units as the input. End points are duplicated so the
    curve starts and ends exactly at the first and last control points.
    """
    p = np.asarray(points, dtype=float)
    if len(p) < 2:
        raise ValueError('catmull_rom() needs at least two points')
    ext = np.vstack([p[0], p, p[-1]])
    out = [p[0]]
    for i in range(1, len(ext) - 2):
        p0, p1, p2, p3 = ext[i - 1], ext[i], ext[i + 1], ext[i + 2]
        for k in range(1, samples_per_segment + 1):
            u = k / samples_per_segment
            u2, u3 = u * u, u * u * u
            out.append(0.5 * ((2 * p1) + (-p0 + p2) * u + (2 * p0 - 5 * p1 + 4 * p2 - p3) * u2
                              + (-p0 + 3 * p1 - 3 * p2 + p3) * u3))
    return np.asarray(out)

    def platoon(path: Sequence[Waypoint], n: int, headway_s: float = 1.0, lateral: float = 0.0,
            jitter_m: float = 0.0, jitter_s: float = 0.0, seed: int = 0,
            lateral_axis: str = 'y') -> list[list[Waypoint]]:

    ``lateral`` staggers alternate riders sideways (metres along ``lateral_axis``); ``jitter_m``
    and ``jitter_s`` add seeded uniform noise to each rider's lateral position and start time so
    the group does not look cloned. The result is deterministic for a given ``seed``.
    """

    out = []
    for i in range(n):
        side = lateral * (1 if i % 2 else 0)
        off = side + (rng.uniform(-jitter_m, jitter_m) if jitter_m else 0.0)
        dt = i * headway_s + (rng.uniform(0, jitter_s) if jitter_s and i else 0.0)
        if lateral_axis == 'y':
            out.append(shift(path, dy=off, dt=dt))
        else:
            out.append(shift(path, dx=off, dt=dt))
    return out

    def path_accelerations(path: Sequence[Waypoint]) -> np.ndarray:
    """Magnitude of the velocity change (m/s^2) at each interior waypoint.

    The velocity change between consecutive segments is divided by the time between segment
    midpoints, so turns contribute their centripetal acceleration.

    for i0, i1 in _runs(dangers):
        lo = max(0, i0 - int(round(LEAD_WINDOW_S * fps)))
        hits = np.flatnonzero(is_alert[lo:i1])
        if len(hits):
            first = lo + int(hits[0])
            while first > 0 and is_alert[first - 1]:
                first -= 1
            lead = float(times[i0] - times[first])
            intervals.append({'start': round(float(times[i0]), 3),
                              'end': round(float(times[i1 - 1]) + dt, 3),
                              'alerted': True, 'lead_s': round(lead, 3)})
        else:
            intervals.append({'start': round(float(times[i0]), 3),
                              'end': round(float(times[i1 - 1]) + dt, 3),
                              'alerted': False, 'lead_s': None})
    dominant = dominant_state(states, fps)
    expected = scenario.get('expected_dominant')
    return {
        'id': scenario.get('id', ''),
        'name': scenario['name'],
        'suite': scenario.get('suite', ''),
        'family': scenario.get('family', ''),
        'duration_s': round(float(tb.get_scenario_duration(scenario)), 3),
        'agents': len(scenario['agents']),
        'danger_s': round(float(np.sum(dangers)) * dt, 3),
        'actionable_s': round(float(np.sum(actionable)) * dt, 3),
        'alert_s': round(float(np.sum(is_alert)) * dt, 3),
        'false_alert_s': round(r['false_pos'] * dt, 3),
        'danger_intervals': intervals,
        'sensitivity': round(float(r['sensitivity']), 4),
        'specificity': round(float(r['specificity']), 4),
        'dominant': dominant,
        'expected': expected,
        'ground_truth': ground_truth_label(scenario, dangers, fps),
        'match': dominant == expected,
        'tags': list(scenario.get('tags', [])),
    }

    def generate(family: str, n: int = 20, seed: int = 0) -> list[dict]:
    """``n`` scenarios of ``family`` (a key of :data:`FAMILIES`), deterministic for ``seed``."""
    if family not in FAMILIES:
        raise KeyError(f'unknown family {family!r}; known: {", ".join(FAMILIES)}')
    if n < 1:
        raise ValueError('n must be at least 1')
    rng = np.random.default_rng(seed)

    def jt() -> float:
        return float(rng.uniform(-0.2, 0.2))

    def js(v: float) -> float:
        return v * float(rng.uniform(0.95, 1.05))

    out: list[dict] = []
    if family == 'crossing-angle':
        for ang in _sweep(30.0, 150.0, n):
            out.append(_base(
                f'Crossing at {ang:.0f} deg',
                f'A pedestrian crosses at {ang:.0f} degrees to the road and reaches the lane '
                f'centre as a {BIKE_SPEED:.0f} m/s cyclist arrives.',
                [ped(1, _cross_through((5.0, 0.0), ang, js(PED_SPEED), T_CONFLICT + jt())),
                 bike(2, _ride(js(BIKE_SPEED), T_CONFLICT + jt()))]))
    elif family == 'cyclist-speed':
        for v in _sweep(3.0, 11.0, n):
            out.append(_base(
                f'Cyclist at {v:.1f} m/s',
                f'A cyclist at {v:.1f} m/s reaches the crosswalk as the pedestrian is mid-lane.',
                [ped(1, _cross_through((5.0, 0.0), 90.0, js(PED_SPEED), T_CONFLICT + jt())),
                 bike(2, _ride(js(v), T_CONFLICT + jt()),
                      model='ebike' if v > 8.5 else 'bicycle')],
                vehicle='ebike' if v > 8.5 else 'bicycle'))
    elif family == 'time-gap':
        for gap in _sweep(-3.0, 3.0, n):
            out.append(_base(
                f'Time gap {gap:+.1f} s',
                f'The cyclist reaches the conflict point {abs(gap):.1f} s '
                f'{"after" if gap >= 0 else "before"} the pedestrian.',
                [ped(1, _cross_through((5.0, 0.0), 90.0, js(PED_SPEED), T_CONFLICT)),
                 bike(2, _ride(js(BIKE_SPEED), T_CONFLICT + gap + 0.5 * jt()))]))
    elif family == 'lateral-offset':
        for off in _sweep(0.0, 5.0, n):
            out.append(_base(
                f'Lateral offset {off:.1f} m',
                f'The cyclist passes {off:.1f} m to the side of the pedestrian at the moment '
                f'of crossing.',
                [ped(1, _cross_through((5.0, 0.0), 90.0, js(PED_SPEED), T_CONFLICT + jt())),
                 bike(2, _ride(js(BIKE_SPEED), T_CONFLICT, y=off))]))
    elif family == 'group-size':
        for k in range(1, n + 1):
            lead = _cross_through((5.0, 0.0), 90.0, PED_SPEED, T_CONFLICT)
            offsets = [(float(rng.uniform(-1.2, 1.2)), -0.8 * (i // 2), 0.15 * i + abs(0.1 * jt()))
                       for i in range(k)]
            out.append(_base(
                f'Group of {k}',
                f'{k} pedestrian{"s" if k > 1 else ""} cross together into the path of one '
                f'cyclist.',
                [*[ped(i + 1, p) for i, p in enumerate(group(lead, offsets))],
                 bike(k + 1, _ride(js(BIKE_SPEED), T_CONFLICT + jt()))]))
    else:
        for _ in range(n):
            n_p, n_b = int(rng.integers(1, 5)), int(rng.integers(1, 4))
            agents = []
            for i in range(n_p):
                agents.append(ped(i + 1, _cross_through(
                    (float(rng.uniform(2, 8)), float(rng.uniform(-2, 2))),
                    float(rng.uniform(55, 125)) + (180.0 if rng.random() < 0.5 else 0.0),
                    float(rng.uniform(1.1, 1.6)), float(rng.uniform(4, 9)))))

                f'{n_p} pedestrian{"s" if n_p > 1 else ""} and {n_b} '
                f'cyclist{"s" if n_b > 1 else ""} at random times, angles and speeds.',
                agents))
    return _finish(out, family, seed)

    def validate_scenario(s: dict) -> list[str]:
    """Problems found in one scenario (empty list when valid)."""
    problems: list[str] = []
    label = s.get('id') or s.get('name') or '<unnamed>'

    def bad(msg: str) -> None:
        problems.append(f'{label}: {msg}')

    if not isinstance(s.get('name'), str) or not s.get('name'):
        bad('missing name')
    if s.get('expected_dominant') not in STATES:
        bad(f"expected_dominant {s.get('expected_dominant')!r} not one of {STATES}")
    if 'vehicle' in s and s['vehicle'] not in VEHICLES:
        bad(f"unknown vehicle {s['vehicle']!r}")
    if 'suite' in s and not isinstance(s['suite'], str):
        bad('suite must be a string')
    if 'family' in s and s['family'] not in _known_families():
        bad(f"unknown family {s['family']!r}")
    if 'tags' in s and not all(isinstance(t, str) for t in s['tags']):
        bad('tags must be strings')
    weather = (s.get('render') or {}).get('weather')
    if weather is not None and weather not in WEATHER:
        bad(f'unknown weather preset {weather!r}')
    agents = s.get('agents')
    if not isinstance(agents, list) or not agents:
        bad('needs a non-empty agents list')
        return problems
    coarse = s.get('suite') == 'conformance'
    ids = [a.get('id') for a in agents]
    if len(set(ids)) != len(ids):
        bad(f'duplicate agent ids {ids}')
    for a in agents:
        who = f"agent {a.get('id')}"
        if a.get('type') not in AGENT_TYPES:
            bad(f"{who}: type {a.get('type')!r} not one of {AGENT_TYPES}")
            continue
        model = a.get('model')
        allowed = PERSON_MODELS if a['type'] == 'person' else BIKE_MODELS
        if model is not None and model not in allowed:
            bad(f'{who}: model {model!r} not valid for a {a["type"]}')
            continue
        path = a.get('path')
        if not isinstance(path, list) or len(path) < 2:
            bad(f'{who}: path needs at least two waypoints')
            continue
        if not all(isinstance(p, dict) and all(_finite(p.get(k)) for k in 'xyt') for p in path):
            bad(f'{who}: every waypoint needs finite x, y, t')
            continue
        ts = [p['t'] for p in path]
        if any(t1 <= t0 for t0, t1 in zip(ts, ts[1:])):
            bad(f'{who}: waypoint times must be strictly increasing')
            continue
        if ts[0] < 0:
            bad(f'{who}: negative start time')
        vmax, amax = LIMITS[(a['type'], model)]
        speeds = path_speeds(path)
        if speeds.max() > vmax + 1e-6:
            bad(f'{who}: speed {speeds.max():.2f} m/s exceeds {vmax} m/s for '
                f'{model or a["type"]}')
        acc = path_accelerations(path)
        limit = amax * (COARSE_ACCEL_FACTOR if coarse else 1.0)
        if len(acc) and acc.max() > limit + 1e-6:
            bad(f'{who}: acceleration {acc.max():.2f} m/s^2 exceeds {limit} m/s^2')
    return problems

    """Validate every scenario and check that ids and names are unique among them."""
    problems: list[str] = []
    seen_ids: dict[str, str] = {}
    seen_names: dict[str, str] = {}
    for s in scenarios:
        problems.extend(validate_scenario(s))
        where = f"{s.get('suite', '?')}"
        sid = s.get('id')
        if not sid:
            problems.append(f"{s.get('name')}: missing id")
        elif sid in seen_ids:
            problems.append(f'duplicate id {sid!r} in {seen_ids[sid]} and {where}')
        else:
            seen_ids[sid] = where
        name = s.get('name')
        if name in seen_names:
            problems.append(f'duplicate name {name!r} in {seen_names[name]} and {where}')
        else:
            seen_names[name] = where
    return problems
